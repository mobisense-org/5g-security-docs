## Master Table (2023-2026)

Most cells in each row link to the **attack's detail page** (so you can click ~anywhere in the row to open it). The **Paper** column links to the original paper or vendor disclosure. The **Code** column links to a public repo when one exists.

**Detection key**

- `RF AI` — detectable from a passive air-interface sniffer + ML classifier (no operator access required).
- `BS Logs` — requires gNB / 5G-core / SBA telemetry to detect; not visible on the air.
- `Both` — observable from either side.
- `Neither` — attacker is itself passive or operates out-of-band; effectively undetectable, only mitigatable.

**Threat model key**

- `Passive RF` — sniffer in radio coverage, no transmission.
- `Active RF` — transmits on the air (overshadowing, injection, jammer) but no full fake cell.
- `Rogue gNB` — full fake base station that the UE attaches to.
- `Relay gNB` — two SDRs sitting between victim UE and a real gNB.
- `Rogue UE` — attacker-controlled UE sending crafted uplink (NAS/RRC) messages.
- `SBA` — needs IP-level reach into the operator's 5G core / Service-Based Architecture.
- `Backhaul` — needs reach to transport between RAN and core (N3/N9, GTP).
- `Internet` — public-Internet attacker (only when operator misconfigures exposure).
- `Physical` — needs hardware access to a device (femtocell, CPE, modem).

    function venueVal(r)  { return (r.cells[cols.venue] || {}).textContent?.trim().toLowerCase() || ""; }
    function catVal(r)    { return ((r.cells[cols.cat] || r.cells[cols.category] || {}).textContent?.trim() || "").toLowerCase(); }
    function codeVal(r)   { return (r.cells[cols.code] || {}).textContent?.trim().toLowerCase() || ""; }
    function replVal(r)   { const t = (r.cells[cols.repl] || {}).textContent?.trim() || ""; const m = t.match(/(\\d+)/); return m ? parseInt(m[1]) : 0; }
    function isCandidate(r) {
      const d = detectVal(r);
      const cv = codeVal(r);
      const isNoCode = !cv || cv === "no" || cv === "—" || cv === "-" || cv === "not yet" || cv === "partial";
      return replVal(r) >= 4 && (d === "rf ai" || d === "both") && !isNoCode;
    }

    // Tag each row with data-threat, data-detect, and class="candidate" for CSS targeting
    rows.forEach(r => {
      const slug = s => s.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, "");
      const tc = r.cells[cols.threat];
      if (tc) tc.setAttribute("data-threat", slug(tc.textContent.trim()));
      const dc = r.cells[cols.detect];
      if (dc) dc.setAttribute("data-detect", slug(dc.textContent.trim()));
      if (isCandidate(r)) r.classList.add("candidate");
    });

    function apply() {
      const q = toolbar.querySelector("#atk-q").value.trim().toLowerCase();
      const fd = toolbar.querySelector("#atk-detect").value.toLowerCase();
      const ft = toolbar.querySelector("#atk-threat").value.toLowerCase();
      const fv = toolbar.querySelector("#atk-venue").value.toLowerCase();
      const fc = toolbar.querySelector("#atk-cat").value;
      const fk = toolbar.querySelector("#atk-code").value;
      const fr = toolbar.querySelector("#atk-repl").value;

      let visible = 0;
      rows.forEach(r => {
        const text = r.textContent.toLowerCase();
        const cv = codeVal(r);
        const isNoCode = !cv || cv === "no" || cv === "—" || cv === "-" || cv === "not yet";
        let ok = true;
        if (candidatesOnly && !r.classList.contains("candidate")) ok = false;
        if (q && !text.includes(q)) ok = false;
        if (fd && detectVal(r) !== fd) ok = false;
        if (ft && threatVal(r) !== ft) ok = false;
        if (fv && venueVal(r) !== fv) ok = false;
        if (fc && catVal(r) !== fc) ok = false;
        if (fk === "yes" && isNoCode) ok = false;
        if (fk === "no" && !isNoCode) ok = false;
        if (fr && replVal(r) < parseInt(fr)) ok = false;
        r.style.display = ok ? "" : "none";
        if (ok) visible++;
      });

      if (sortCol !== null && sortDir !== 0) {
        const visibleRows = rows.filter(r => r.style.display !== "none");
        visibleRows.sort((a, b) => {
          const av = (a.cells[sortCol] || {}).textContent?.trim() || "";
          const bv = (b.cells[sortCol] || {}).textContent?.trim() || "";
          const an = parseFloat(av), bn = parseFloat(bv);
          if (!isNaN(an) && !isNaN(bn)) return (an - bn) * sortDir;
          return av.localeCompare(bv) * sortDir;
        });
        visibleRows.forEach(r => tbody.appendChild(r));
      }

      toolbar.querySelector("#atk-stats").textContent = visible + " of " + rows.length + " attacks";
    }

    headers.forEach((th, i) => {
      th.addEventListener("click", () => {
        if (sortCol === i) {
          sortDir = sortDir === 1 ? -1 : sortDir === -1 ? 0 : 1;
          if (sortDir === 0) sortCol = null;
        } else { sortCol = i; sortDir = 1; }
        headers.forEach(h => h.classList.remove("sort-asc", "sort-desc"));
        if (sortDir === 1) th.classList.add("sort-asc");
        if (sortDir === -1) th.classList.add("sort-desc");
        apply();
      });
    });

    toolbar.addEventListener("input", apply);
    toolbar.addEventListener("change", apply);
    toolbar.querySelector("#atk-candidates").addEventListener("click", e => {
      candidatesOnly = !candidatesOnly;
      e.currentTarget.classList.toggle("active", candidatesOnly);
      apply();
    });
    toolbar.querySelector("#atk-reset").addEventListener("click", () => {
      toolbar.querySelectorAll("input, select").forEach(el => el.value = "");
      candidatesOnly = false;
      toolbar.querySelector("#atk-candidates").classList.remove("active");
      sortCol = null; sortDir = 0;
      headers.forEach(h => h.classList.remove("sort-asc", "sort-desc"));
      apply();
    });

    apply();
    return true;
  }

  // Retry until table is in the DOM (handles SPA / late hydration)
  let tries = 0;
  const t = setInterval(() => {
    if (enhance() || ++tries > 40) clearInterval(t);
  }, 200);
})();
